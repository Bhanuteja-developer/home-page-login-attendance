/**
 * Google Apps Script Backend (Code.gs)
 * Deploy as Web App -> Execute as: Me -> Access: Anyone
 */
const SHEET_NAME = 'Vehicle_ODO_Logs';
const DRIVE_FOLDER_NAME = 'Vehicle_Tracker_Images';

function doGet() {
  return HtmlService.createHtmlOutputFromFile('index')
    .setTitle('Vehicle Tracker - Smart ODO Management')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no');
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const result = saveVehicleLogToCloud(payload);
    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function saveVehicleLogToCloud(data) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      sheet = ss.insertSheet(SHEET_NAME);
      sheet.appendRow([
        'Log_ID', 'ISO_Timestamp', 'Date_Label', 'Mode', 'Reading_Type',
        'Odometer_Km', 'Fuel_Liters', 'Amount_INR', 'GPS_Address',
        'GPS_Coords', 'Image_Drive_URL', 'User_Email'
      ]);
    }
    let imageUrl = '';
    if (data.imageBase64 && data.imageBase64.startsWith('data:image/')) {
      const folders = DriveApp.getFoldersByName(DRIVE_FOLDER_NAME);
      const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(DRIVE_FOLDER_NAME);
      const parts = data.imageBase64.split(',');
      const mime = parts[0].match(/:(.*?);/)[1];
      const bytes = Utilities.base64Decode(parts[1]);
      const blob = Utilities.newBlob(bytes, mime, `ODO_${data.readingType}_${Date.now()}.jpg`);
      const file = folder.createFile(blob);
      imageUrl = file.getUrl();
    }
    sheet.appendRow([
      Utilities.getUuid(),
      new Date().toISOString(),
      String(data.dateLabel || '').slice(0, 40),
      String(data.mode || '').slice(0, 30),
      String(data.readingType || '').slice(0, 40),
      Number(data.odometerKm) || 0,
      Number(data.fuelLiters) || 0,
      Number(data.amountPaid) || 0,
      String(data.gpsAddress || '').slice(0, 200),
      String(data.gpsCoords || '').slice(0, 60),
      imageUrl,
      Session.getActiveUser().getEmail() || 'anonymous'
    ]);
    return { ok: true, imageUrl: imageUrl, timestamp: new Date().toISOString() };
  } finally {
    lock.releaseLock();
  }
}
